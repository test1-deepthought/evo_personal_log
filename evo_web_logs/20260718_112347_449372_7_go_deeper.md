%% Conclusion 1: Each framework's complete answer across all 4 meaning-senses
conclusion(framework_answer(Framework, Sense, Answer)) :-
    framework(Framework, _),
    sense_of_meaning(Sense, _),
    framework_conclusion(Framework, Sense, Answer).

%% Conclusion 2: Frameworks grouped by grounding type
conclusion(objective_grounded(Frameworks)) :-
    findall(F, (framework(F, _),
                framework_type(F, Type),
                sub_atom(Type, 0, _, _, 'Objective')),
            Frameworks).

conclusion(subjective_grounded(Frameworks)) :-
    findall(F, (framework(F, _),
                framework_type(F, Type),
                sub_atom(Type, 0, _, _, 'Subjective')
                ; framework_type(F, Type), sub_atom(Type, 0, _, _, 'Descriptive')
                ; framework_type(F, Type), sub_atom(Type, 0, _, _, 'Pragmatic')),
            Frameworks).

%% Conclusion 3: Frameworks that affirm vs deny meaning
conclusion(meaning_affirmed(Frameworks)) :-
    findall(F, framework(F, _), All),
    findall(F, (member(F, All), framework_type(F, 'Objective denial — meaning is objectively absent.')), Deniers),
    findall(F, (member(F, All), framework_type(F, 'Descriptive — meaning is explained away as an evolutionary artifact.')), Explainers),
    findall(F, (member(F, All),
                \+ member(F, Deniers),
                \+ member(F, Explainers)),
            Frameworks).

conclusion(meaning_denied(Frameworks)) :-
    findall(F, (framework_type(F, 'Objective denial — meaning is objectively absent.')
             ;  framework_type(F, 'Descriptive — meaning is explained away as an evolutionary artifact.')),
            Frameworks).
